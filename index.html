def f(x): 
    return -x**2 + 4*x 
 
def hill_climbing(start, step_size, max_iterations): 
    current_solution = start 
    current_value = f(current_solution) 
     
    for _ in range(max_iterations): 
        neighbor_left = current_solution - step_size 
        neighbor_right = current_solution + step_size 
         
        neighbor_left_value = f(neighbor_left) 
        neighbor_right_value = f(neighbor_right) 
         
        if neighbor_left_value > current_value: 
            current_solution = neighbor_left 
            current_value = neighbor_left_value 
        elif neighbor_right_value > current_value: 
            current_solution = neighbor_right 
            current_value = neighbor_right_value 
        else: 
            break 
 
    return current_solution, current_value

import math 
PLAYER_X, PLAYER_O, EMPTY = "X", "O", " " 
 
def check_winner(board, player): 
    win_combinations = [(0,1,2), (3,4,5), (6,7,8), (0,3,6), (1,4,7), (2,5,8), (0,4,8), (2,4,6)] 
    return any(all(board[i] == player for i in combo) for combo in win_combinations) 
 
def is_board_full(board): 
    return all(spot != EMPTY for spot in board) 
 
def minimax(board, depth, alpha, beta, is_maximizing): 
    if check_winner(board, PLAYER_X): return -10 + depth 
    if check_winner(board, PLAYER_O): return 10 - depth 
    if is_board_full(board): return 0 
 
    if is_maximizing: 
        max_eval = -math.inf 
        for i in range(9): 
            if board[i] == EMPTY: 
                board[i] = PLAYER_O 
                eval = minimax(board, depth + 1, alpha, beta, False) 
                board[i] = EMPTY 
                max_eval = max(max_eval, eval) 
                alpha = max(alpha, eval) 
                if beta <= alpha: break 
        return max_eval 
    else: 
        min_eval = math.inf 
        for i in range(9): 
            if board[i] == EMPTY: 
                board[i] = PLAYER_X 
                eval = minimax(board, depth + 1, alpha, beta, True) 
                board[i] = EMPTY 
                min_eval = min(min_eval, eval) 
                beta = min(beta, eval) 
                if beta <= alpha: break 
        return min_eval 
 
def best_move(board): 
    best_val = -math.inf 
    move = -1 
    for i in range(9): 
        if board[i] == EMPTY: 
            board[i] = PLAYER_O 
            move_val = minimax(board, 0, -math.inf, math.inf, False) 
            board[i] = EMPTY 
            if move_val > best_val: 
                best_val = move_val 
                move = i 
    return move 
 
def display(board):

        print(f"{board[i]} | {board[i+1]} | {board[i+2]}")

import heapq 
 
GOAL_STATE = ((1, 2, 3), (8, 0, 4), (7, 6, 5)) 
MOVES = [(-1, 0), (1, 0), (0, -1), (0, 1)] 
 
def manhattan_distance(state): 
    distance = 0 
    for r in range(3): 
        for c in range(3): 
            value = state[r][c] 
            if value != 0: 
                goal_r, goal_c = (value - 1) // 3, (value - 1) % 3 
                distance += abs(r - goal_r) + abs(c - goal_c) 
    return distance 
 
def is_valid_move(x, y): 
    return 0 <= x < 3 and 0 <= y < 3 
 
def get_neighbors(state): 
    neighbors = [] 
    x, y = next((r, c) for r in range(3) for c in range(3) if state[r][c] == 0) 
    for dx, dy in MOVES: 
        new_x, new_y = x + dx, y + dy 
        if is_valid_move(new_x, new_y): 
            new_state = [list(row) for row in state] 
            new_state[x][y], new_state[new_x][new_y] = new_state[new_x][new_y], new_state[x][y] 
            neighbors.append(tuple(tuple(row) for row in new_state)) 
    return neighbors 
 
def a_star(start_state): 
    start_state = tuple(tuple(row) for row in start_state) 
    if start_state == GOAL_STATE: 
        return [] 
    open_list = [(manhattan_distance(start_state), 0, start_state, [])] 
    visited = set() 
 
    while open_list: 
        f, g, state, path = heapq.heappop(open_list) 
        if state == GOAL_STATE: 
            return path 
        if state in visited: 
            continue 
        visited.add(state) 
        for neighbor in get_neighbors(state): 
            if neighbor not in visited: 
                f = g + 1 + manhattan_distance(neighbor) 
                heapq.heappush(open_list, (f, g + 1, neighbor, path + [neighbor])) 
    return None

def dfs(graph, node, goal, depth, visited): 
    if depth == 0: 
        if node == goal: 
            return [node] 
        return None 
    if depth > 0: 
        visited.add(node) 
        for neighbor in graph.get(node, []): 
            if neighbor not in visited: 
                path = dfs(graph, neighbor, goal, depth - 1, visited) 
                if path:

    return None

def hill_climbing_maximize(func, x_start, step_size, precision): 
    current_x = x_start 
    while True: 
        x_left = current_x - step_size 
        x_right = current_x + step_size 
         
        if func(x_left) > func(current_x): 
            current_x = x_left 
        elif func(x_right) > func(current_x):

            break

import heapq 
class State: 
    def __init__(self, width, height, remaining_objects, placed_objects): 
        self.width = width  # Room width 
        self.height = height  # Room height 
        self.remaining_objects = remaining_objects  # Objects yet to be placed 
        self.placed_objects = placed_objects  # Objects already placed in the room 
 
    # Less than comparison for priority queue based on heuristic 
    def __lt__(self, other): 
        return self.heuristic() < other.heuristic() 
 
    # Heuristic function to guide the A* search 
    def heuristic(self): 
        # We use the remaining space in the room as the heuristic (minimize remaining space) 
        placed_area = sum(obj[0] * obj[1] for obj in self.placed_objects)  # Area covered by placed 
objects 
        total_area = self.width * self.height  # Total room area 
        remaining_area = total_area - placed_area  # Remaining space in the room 
        return remaining_area 
 
    # Goal check: all objects are placed 
    def is_goal(self): 
        return len(self.remaining_objects) == 0 
 
    # Function to try placing an object and return the new state 
    def place_object(self, obj): 
        new_remaining_objects = [o for o in self.remaining_objects if o != obj] 
        new_placed_objects = self.placed_objects + [obj] 
        return State(self.width, self.height, new_remaining_objects, new_placed_objects) 
 
# A* search function to arrange the objects 
def a_star(room_width, room_height, rectangular_objects, square_objects): 
    initial_state = State(room_width, room_height, rectangular_objects + square_objects, []) 
    open_set = [] 
    heapq.heappush(open_set, initial_state) 
 
    while open_set: 
        current_state = heapq.heappop(open_set) 
 
        # If goal state reached (all objects are placed) 
        if current_state.is_goal(): 
            return current_state 
 
        # Try placing each remaining object in the room 
        for obj in current_state.remaining_objects: 
            # Try horizontal placement 
            if obj[0] <= current_state.width and obj[1] <= current_state.height: 
                new_state = current_state.place_object(obj) 
                heapq.heappush(open_set, new_state) 
 
            # Try vertical placement 
            if obj[1] <= current_state.width and obj[0] <= current_state.height: 
                new_state = current_state.place_object(obj) 
                heapq.heappush(open_set, new_state) 
 
    return None

import heapq 
# Define the goal state 
GOAL_STATE = ((1, 2, 3), (8, 0, 4), (7, 6, 5)) 
# Define possible moves (up, down, left, right) 
MOVES = [(-1, 0), (1, 0), (0, -1), (0, 1)]  # (row, col) 
# Manhattan distance heuristic 
def manhattan_distance(state): 
distance = 0 
    for r in range(3): 
        for c in range(3): 
            value = state[r][c] 
            if value != 0: 
                goal_r, goal_c = (value - 1) // 3, (value - 1) % 3 
                distance += abs(r - goal_r) + abs(c - goal_c) 
    return distance 
 
# Check if the move is valid (within bounds) 
def is_valid_move(x, y): 
    return 0 <= x < 3 and 0 <= y < 3 
 
# Get all possible neighbors (states) from the current state 
def get_neighbors(state): 
    neighbors = [] 
    # Find the position of the empty tile (0) 
    x, y = next((r, c) for r in range(3) for c in range(3) if state[r][c] == 0) 
     
    for dx, dy in MOVES: 
        new_x, new_y = x + dx, y + dy 
        if is_valid_move(new_x, new_y): 
            # Make a new state by swapping the empty tile with a neighboring tile 
            new_state = [list(row) for row in state]  # Convert to list of lists to modify 
            new_state[x][y], new_state[new_x][new_y] = new_state[new_x][new_y], new_state[x][y] 
            neighbors.append(tuple(tuple(row) for row in new_state))  # Convert back to tuple of tuples 
    return neighbors 
 
# A* algorithm to solve the 8-puzzle problem 
def a_star(start_state): 
    start_state = tuple(tuple(row) for row in start_state) 
    if start_state == GOAL_STATE: 
        return []  # Already at the goal, no need to move 
 
    open_list = [] 
    heapq.heappush(open_list, (manhattan_distance(start_state), 0, start_state, [])) 
    visited = set() 
 
    while open_list: 
        f, g, state, path = heapq.heappop(open_list) 
        if state == GOAL_STATE: 
            return path  # Return the solution path 
 
        if state in visited: 
            continue 
        visited.add(state) 
 
        for neighbor in get_neighbors(state): 
            if neighbor not in visited: 
                f = g + 1 + manhattan_distance(neighbor) 
                heapq.heappush(open_list, (f, g + 1, neighbor, path + [neighbor])) 
 
    return None  # No solution found 
 
# Function to print a state (puzzle configuration) 
def print_state(state): 
    for row in state: 
        print(" ".join(map(str, row))) 
    print() 
 
# Main function to input and solve the puzzle 
if __name__ == "__main__": 
    # Input the initial state of the puzzle 
    print("Enter the initial state of the 8-puzzle (3x3 matrix, use 0 for the blank space):") 
    start_state = [] 
    for i in range(3): 
        row = list(map(int, input(f"Enter row {i+1} (space-separated): ").split())) 
        start_state.append(row) 
 
    # Solve the puzzle using A* algorithm 
    solution = a_star(start_state) 
 
    if solution: 
        print("\nSolution found!") 
        for state in solution: 
            print_state(state)  # Print each step of the solution path 
    else: 
        print("No solution found.")
